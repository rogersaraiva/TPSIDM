Exercício 1.1 — Contagem por estado
enum class Status {
    OPEN, IN_PROGRESS, CLOSED
}

data class Occurrence(
    val id: Int,
    val status: Status
)

fun List<Occurrence>.countByStatus(): Map<Status, Int> {
    return this
        .groupBy { it.status }
        .mapValues { it.value.size }
}

fun main() {
    val occurrences = listOf(
        Occurrence(1, Status.OPEN),
        Occurrence(2, Status.IN_PROGRESS),
        Occurrence(3, Status.CLOSED)
    )

    println(occurrences.countByStatus())
}


Resultado esperado:
{OPEN=1, IN_PROGRESS=1, CLOSED=1}
